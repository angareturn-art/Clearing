3. KPI 카드 컴포넌트 패턴
// KpiCard.jsx (신규 공용 컴포넌트)
function KpiCard({label, value, total,
  unit='', color='green'}) {
  const pct = total
    ? Math.round(value/total*100) : null;
  return (
    <div className="bg-white rounded-xl p-5
      border border-gray-100 shadow-sm">
      <p className="text-xs font-bold uppercase
        tracking-widest text-gray-400 mb-2">
        {label}
      </p>
      <p className="text-3xl font-black
        text-navy-primary">
        {value.toLocaleString()}{unit}
      </p>
      <div className="h-1 bg-gray-100
        rounded mt-3 overflow-hidden">
        <div style={{width:`${pct}%`}}
          className="h-full bg-green-primary
          rounded transition-all"></div>
      </div>
      {pct !== null && (
        <p className="text-xs font-bold
          text-green-primary mt-2">
          ↑ {pct}% 완료
        </p>
      )}
    </div>
  )
}
