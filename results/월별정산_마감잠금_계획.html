추가할 state / 함수

// 추가할 state
const [closedMonths, setClosedMonths] = useState(new Set());
const [closingInProgress, setClosingInProgress] = useState(false);

// 파생값
const isClosed = closedMonths.has(month);   // 현재 선택 월 마감 여부

// 마감된 월 목록 로드 (월 변경 시마다 갱신)
const fetchClosedMonths = async () => {
  const res = await fetch('/api/monthly-closings', { headers });
  const rows = await res.json();
  setClosedMonths(new Set(rows.map(r => r.month)));
};

// 마감 처리
const doClose = async () => {
  setClosingInProgress(true);
  await fetch('/api/monthly-closings', {
    method: 'POST',
    headers: { ...headers, 'Content-Type': 'application/json' },
    body: JSON.stringify({ month }),
  });
  await fetchClosedMonths();
  setClosingInProgress(false);
};

// 마감 취소
const doUnclose = async () => {
  if (!window.confirm(`${month} 마감을 취소하시겠습니까?`)) return;
  await fetch(`/api/monthly-closings/${month}`, { method: 'DELETE', headers });
  await fetchClosedMonths();
};

엑셀 저장 함수 수정 — 저장 후 마감 제안

const handleExcelDownload = async () => {
  // ... 기존 엑셀 생성 로직 동일 ...

  // ── 추가: 저장 완료 후 미마감이면 마감 제안 ──
  if (!isClosed) {
    const ok = window.confirm(
      `${month} 엑셀 저장이 완료됐습니다.\n\n이 월을 마감하시겠습니까?\n마감 후 해당 월 데이터는 수정·삭제가 잠깁니다.`
    );
    if (ok) await doClose();
  }
};

3 서버 측 — 이미 구현 완료 변경 불필요
항목내용상태
DB 테이블 monthly_closings (site_id, month, closed_at, closed_by) ✅ 완료
마감 조회 API GET /api/monthly-closings — 마감된 월 목록 ✅ 완료
마감 처리 API POST /api/monthly-closings — 월 마감 등록 ✅ 완료
마감 취소 API DELETE /api/monthly-closings/:month ✅ 완료
오일링 기록 잠금 POST/PUT/DELETE /api/records/oiling → 마감 월 차단 ✅ 완료
청소 기록 잠금 POST/PUT/DELETE /api/records/cleaning + 서명 처리 차단 ✅ 완료
하역 기록 잠금 POST/PUT/DELETE /api/records/unloading → 마감 월 차단 ✅ 완료
인건비 기록 잠금 POST/PUT/DELETE /api/personnel → 마감 월 차단 ✅ 완료
비용 기록 잠금 PUT/DELETE /api/costs → 마감 월 차단 ✅ 완료
🔒 잠금 동작 마감된 월의 데이터를 수정·삭제하면 서버가 HTTP 403을 반환합니다.
메시지: "YYYY-MM 은 마감 완료된 월입니다. 수정하려면 마감 취소 후 진행하세요."
